#version 130

in vec2 fragmentPosition;
in vec4 fragmentColor;
in vec2 fragmentUV;

out vec4 color;
uniform float time;
uniform sampler2D myImage;

void main(){
	float PI = 3.14159265359;
	float TWO_PI = 6.28318530718;

	vec2 base_uv = fragmentUV;
	base_uv -= 0.5;
	base_uv *= 2.0;

	float angle = atan(base_uv.y, base_uv.x) / (2.0 * PI);
	float dist = length(vec2(base_uv.x, base_uv.y));

	vec2 polar_uv = vec2(angle, dist);

	float wave = sin(10.0 * polar_uv.y - time * 3.0) * 0.05;

	vec2 uv_with_wave = polar_uv;
    uv_with_wave.y += wave;

	base_uv = vec2(cos(uv_with_wave.x * TWO_PI), sin(uv_with_wave.x * TWO_PI)) * uv_with_wave.y;

    base_uv /= 2.0;
    base_uv += 0.5;

    vec4 textureColor = texture(myImage, base_uv);
    color = textureColor;

}
